function main(config) {

  // ============================================================
  // 0. 节点来源识别：同时支持静态 proxies 与动态 proxy-providers
  // ============================================================
  let allProxies = (config.proxies || []).map(p => p.name);
  const providerNames = config["proxy-providers"] ? Object.keys(config["proxy-providers"]) : [];

  // 静态节点的 AI 友好节点筛选（防备纯静态配置场景）
  const aiFriendlyProxies = allProxies.filter(name =>
    /新加坡|SG|Singapore|美国|US|USA|United States|日本|JP|Japan|台湾|TW|Taiwan|台灣|英国|UK|United Kingdom|Britain|韩国|KR|Korea|Seoul|首尔|首爾|加拿大|CA|Canada|Toronto|Vancouver|多伦多|温哥华|溫哥華|Tokyo|Osaka|London|LA|NY/i.test(name) &&
    !/香港|HK|HongKong|中国|CN|China/i.test(name)
  );

  // ============================================================
  // 1. 纯净节点分组：彻底抛弃机场自带分组，只保留你的核心双组
  // ============================================================
  
  // 基础全能代理组
  const proxyGroupDef = {
    name: "My-Proxy",
    type: "url-test",
    url: "http://www.gstatic.com/generate_204",
    interval: 300,
    tolerance: 50,
    lazy: true
  };

  if (providerNames.length > 0) {
    proxyGroupDef.use = providerNames;
  } else {
    proxyGroupDef.proxies = allProxies.length > 0 ? allProxies : ["DIRECT"];
  }

  // AI 专用代理组（基于 provider 或 static 的精准过滤）
  const aiGroupDef = {
    name: "🤖 My-AI",
    type: "url-test",
    url: "https://www.google.com/generate_204",
    interval: 600,
    tolerance: 50,
    lazy: true
  };

  if (providerNames.length > 0) {
    aiGroupDef.use = providerNames;
    aiGroupDef.filter = "(?i)(新加坡|SG|Singapore|美国|US|USA|United States|日本|JP|Japan|台湾|TW|Taiwan|台灣|英国|UK|United Kingdom|Britain|韩国|KR|Korea|Seoul|首尔|首爾|加拿大|CA|Canada|Toronto|Vancouver|多伦多|温哥华|溫哥華|Tokyo|Osaka|London|LA|NY)";
    aiGroupDef["exclude-filter"] = "(?i)(香港|HK|HongKong|中国|CN|China)";
  } else {
    aiGroupDef.proxies = aiFriendlyProxies.length > 0 ? aiFriendlyProxies : (allProxies.length > 0 ? allProxies : ["DIRECT"]);
  }

  // 绝对清爽：直接覆盖，不留任何机场自带的冗余策略组
  config["proxy-groups"] = [proxyGroupDef, aiGroupDef];

  // ============================================================
  // 2. 基础设施直连域名
  // ============================================================
  const infraDirectDomains = [
    "gitee.com",
    "gh-proxy.com",
    "cdn.jsdelivr.net"
  ];

  // ============================================================
  // 3. DNS：内核级规则集直通防护架构
  // ============================================================
  if (!config.dns) config.dns = {};
  config.dns.enable = true;
  config.dns["enhanced-mode"] = "fake-ip";
  config.dns["respect-rules"] = true;  

  config.dns["fake-ip-filter"] = [
    "rule-set:my-direct",
    "rule-set:my-ai",
    "rule-set:direct",
    "rule-set:apple",
    "rule-set:icloud",
    "*.lan",
    "*.local",
    "time.*.com",
    "ntp.*.com",
    "*.msftconnecttest.com",
    "*.srv.nintendo.net",
    "*.pool.ntp.org",
    "*.ntp.org",
    "*.battlenet.com.cn",
    "*.battlenet.com",
    "*.blizzard.com",
    "*.steamcontent.com",
    "*.steamserver.net",
    "*.music.163.com",
    "*.google.com",
    "*.googleapis.com",
    "*.gstatic.com",
    "*.recaptcha.net",
    "*.googleusercontent.com",
    "accounts.google.com",
    "accounts.youtube.com",
    "drive.google.com",
    ...infraDirectDomains.flatMap(d => [d, `*.${d}`])
  ];

  const chinaDoh = [
    "https://dns.alidns.com/dns-query",
    "https://doh.pub/dns-query"
  ];

  config.dns["proxy-server-nameserver"] = chinaDoh;
  config.dns["nameserver"] = chinaDoh;

  config.dns["fallback"] = [
    "https://dns.google/dns-query#My-Proxy",
    "https://1.1.1.1/dns-query#My-Proxy"
  ];

  config.dns["fallback-filter"] = {
    geoip: true,
    "geoip-code": "CN",
    ipcidr: ["240.0.0.0/4"],
    geosite: ["gfw"]
  };

  config.dns["nameserver-policy"] = {
    "geosite:cn": chinaDoh,
    "rule-set:my-direct": chinaDoh,
    "rule-set:my-ai": chinaDoh,
    "rule-set:direct": chinaDoh
  };

  infraDirectDomains.forEach(d => {
    config.dns["nameserver-policy"][d] = chinaDoh;
    config.dns["nameserver-policy"][`*.${d}`] = chinaDoh;
  });

  config.dns["ipv6"] = false;

  config["geodata-mode"] = true;
  config["geodata-loader"] = "standard";
  config["geox-url"] = {
    geosite: "https://cdn.jsdelivr.net/gh/Loyalsoldier/geoip@release/geosite.dat",
    geoip: "https://cdn.jsdelivr.net/gh/Loyalsoldier/geoip@release/geoip.dat"
  };

  // ============================================================
  // 4. Rule Providers
  // ============================================================
  const providers = {
    reject: "domain",
    icloud: "domain",
    apple: "domain",
    google: "domain",
    proxy: "domain",
    direct: "domain",
    private: "domain",
    gfw: "domain",
    tldnotcn: "tld-not-cn",
    lancidr: "ipcidr",
    applications: "classical"
  };

  config["rule-providers"] = config["rule-providers"] || {};

  for (let name in providers) {
    const file = providers[name] === "tld-not-cn" ? "tld-not-cn" : name;
    config["rule-providers"][name] = {
      type: "http",
      behavior: providers[name] === "tld-not-cn" ? "domain" : providers[name],
      url: `https://cdn.jsdelivr.net/gh/Loyalsoldier/clash-rules@release/${file}.txt`,
      path: `./ruleset/${name}.yaml`,
      interval: 86400
    };
  }

  config["rule-providers"]["my-direct"] = {
    type: "http",
    behavior: "domain",
    url: "https://v4.gh-proxy.org/https://raw.githubusercontent.com/child9527/clash-latest/main/clash_rules/my-direct.yaml",
    path: "./ruleset/my-direct.yaml",
    interval: 86400
  };

  config["rule-providers"]["my-ai"] = {
    type: "http",
    behavior: "domain",
    url: "https://v4.gh-proxy.org/https://raw.githubusercontent.com/child9527/clash-latest/main/clash_rules/my-ai.yaml",
    path: "./ruleset/my-ai.yaml",
    interval: 86400
  };

  // ============================================================
  // 5. 分流规则（全套完全自主掌控）
  // ============================================================
  config.rules = [
    // 0. 基础设施直连
    ...infraDirectDomains.map(d => `DOMAIN-SUFFIX,${d},DIRECT`),

    // 1. 核心 AI 服务域名硬编码
    "DOMAIN-SUFFIX,openai.com,🤖 My-AI",
    "DOMAIN-SUFFIX,chatgpt.com,🤖 My-AI",
    "DOMAIN-SUFFIX,anthropic.com,🤖 My-AI",
    "DOMAIN-SUFFIX,claude.ai,🤖 My-AI",
    "DOMAIN-SUFFIX,x.ai,🤖 My-AI",
    "DOMAIN-SUFFIX,grok.com,🤖 My-AI",
    "DOMAIN-SUFFIX,perplexity.ai,🤖 My-AI",
    "DOMAIN-SUFFIX,aistudio.google.com,🤖 My-AI",

    // 2. 个人自定义规则集
    "RULE-SET,my-direct,DIRECT",
    "RULE-SET,my-ai,🤖 My-AI",

    // 3. 基础防骚扰与网络隔离规则
    "RULE-SET,private,DIRECT",
    "RULE-SET,reject,REJECT",
    "RULE-SET,applications,DIRECT",

    // 4. 通用直连规则集与国内优先判定
    "GEOSITE,CN,DIRECT",
    "GEOIP,CN,DIRECT",
    "RULE-SET,direct,DIRECT",
    "RULE-SET,icloud,DIRECT",
    "RULE-SET,apple,DIRECT",

    // 5. 常用代理服务规则集
    "RULE-SET,google,My-Proxy",
    "RULE-SET,proxy,My-Proxy",
    "RULE-SET,gfw,My-Proxy",
    "RULE-SET,tldnotcn,My-Proxy",

    // 6. 最终兜底
    "MATCH,My-Proxy"
  ];

  return config;
}
