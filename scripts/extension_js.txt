  // ============================================================

    ].join(""),
    "gi"
  );

  const HK_TW_MO =
    /台湾|台灣|台北|台版|taiwan|taipei|\btpe\b|\btw\b|香港|港版|hong\s?kong|\bhk\b|澳门|macao|macau/i;

  const AI_FRIENDLY = new RegExp(
    [
      // 亚洲
      "新加坡|狮城|singapore|\\bsg\\b",
      "|日本|东京|大阪|横滨|福冈|札幌|japan|tokyo|osaka|yokohama|fukuoka|sapporo|\\bjp\\b",
      "|韩国|首尔|首爾|釜山|korea|seoul|busan|\\bkr\\b",
      "|马来西亚|吉隆坡|malaysia|kuala\\s?lumpur",

      "|奥地利|维也纳|austria|vienna",
      "|比利时|布鲁塞尔|belgium|brussels",

      "|塞尔维亚|贝尔格莱德|serbia|belgrade",

      // 中东 / 非洲 / 拉美 / 大洋洲
      "|阿联酋|迪拜|\\buae\\b|dubai",

    ].join(""),

  );

  ].join("");

  const BLOCKED = new RegExp("(?:" + RESTRICTED_SOURCE + ")|(?:" + MAINLAND_SOURCE + ")", "i");
  const RESTRICTED_ONLY = new RegExp("(?:" + RESTRICTED_SOURCE + ")", "i");

  function classify(rawName) {
    const geo = String(rawName || "").replace(LINE_NOISE, " ");
    if (BLOCKED.test(geo)) {
      if (RESTRICTED_ONLY.test(geo)) return "restricted";
      if (HK_TW_MO.test(geo)) return "ai";
      return "mainland";
    }
    if (HK_TW_MO.test(geo)) return "ai";
    if (AI_FRIENDLY.test(geo)) return "ai";
    return "unknown";
  }

  const buckets = { ai: [], mainland: [], restricted: [], unknown: [] };
  allProxies.forEach(n => buckets[classify(n)].push(n));

  const premiumScore = n =>
    (/原生|解锁|家宽|住宅|静态|native|residential|unlocked|流媒体/i.test(n) ? 1 : 0);

  const aiMembers = buckets.ai
    .slice()
    .sort((a, b) => premiumScore(b) - premiumScore(a))
    .concat(buckets.unknown);

  const aiFriendlyProxies = aiMembers.length > 0 ? aiMembers : allProxies;

  config["proxy-groups"] = [
    {
      name: MAIN_GROUP,
      type: "url-test",
      proxies: allProxies.length > 0 ? allProxies : ["DIRECT"],
      url: "http://www.gstatic.com/generate_204",
      interval: 300,
      tolerance: 50,
      lazy: true
    },
    {
      name: AI_GROUP,
      type: "url-test",
      proxies: aiFriendlyProxies.length > 0 ? aiFriendlyProxies : ["DIRECT"],
      url: "https://www.google.com/generate_204",
      interval: 600,
      tolerance: 50,
      lazy: true
    }
  ];

  // ============================================================
  // 2. 基础设施直连域名
  // ============================================================
  const infraDirectDomains = ["gitee.com", "gh-proxy.com", "cdn.jsdelivr.net"];

  // 3. DNS：采用 Mihomo 原生标准通配符写法的 fake-ip-filter
  // ============================================================
  if (!config.dns) config.dns = {};
  config.dns.enable = true;
  config.dns["enhanced-mode"] = "fake-ip";
  config.dns["respect-rules"] = true;

  config.dns["fake-ip-filter"] = [
    "rule-set:my-direct",
    "rule-set:my-ai",
    "rule-set:direct",
    "rule-set:apple",
    "rule-set:icloud",
    "*.lan",
    "*.local",
    "time.*.com",
    "ntp.*.com",
    "*.msftconnecttest.com",
    "*.srv.nintendo.net",
    "*.pool.ntp.org",
    "*.ntp.org",
    "*.battlenet.com.cn",
    "*.battlenet.com",
    "*.blizzard.com",
    "*.steamcontent.com",
    "*.steamserver.net",
    "*.music.163.com",
    "*.google.com",
    "*.googleapis.com",
    "*.gstatic.com",
    "*.recaptcha.net",
    "*.googleusercontent.com",
    "accounts.google.com",
    "accounts.youtube.com",
    "drive.google.com",
    ...infraDirectDomains.flatMap(d => [d, `*.${d}`])
  ];

  const chinaDoh = [
    "https://dns.alidns.com/dns-query",
    "https://doh.pub/dns-query"
  ];

  config.dns["proxy-server-nameserver"] = chinaDoh;
  config.dns["nameserver"] = chinaDoh;

  config.dns["fallback"] = [
    "https://dns.google/dns-query#" + MAIN_GROUP,
    "https://1.1.1.1/dns-query#" + MAIN_GROUP
  ];

  config.dns["fallback-filter"] = {
    geoip: true,
    "geoip-code": "CN",
    ipcidr: ["240.0.0.0/4"]
  };

  config.dns["nameserver-policy"] = {
    "geosite:cn": chinaDoh,
    "rule-set:my-direct": chinaDoh,
    "rule-set:my-ai": chinaDoh,
    "rule-set:direct": chinaDoh
  };

  infraDirectDomains.forEach(d => {
    config.dns["nameserver-policy"][d] = chinaDoh;
    config.dns["nameserver-policy"]["*." + d] = chinaDoh;
  });

  config.dns["ipv6"] = false;

  config["geodata-mode"] = true;
  config["geodata-loader"] = "standard";
  config["geox-url"] = {
    geosite: "https://cdn.jsdelivr.net/gh/Loyalsoldier/geoip@release/geosite.dat",
    geoip: "https://cdn.jsdelivr.net/gh/Loyalsoldier/geoip@release/geoip.dat"
  };

  // ============================================================
  // 4. Rule Providers
  // ============================================================
  const providers = {
    reject: "domain",
    icloud: "domain",
    apple: "domain",
    google: "domain",
    proxy: "domain",
    direct: "domain",
    private: "domain",
    gfw: "domain",
    tldnotcn: "tld-not-cn",
    lancidr: "ipcidr",
    applications: "classical"
  };

  config["rule-providers"] = {};

  for (let name in providers) {
    const file = providers[name] === "tld-not-cn" ? "tld-not-cn" : name;
    config["rule-providers"][name] = {
      type: "http",
      behavior: providers[name] === "tld-not-cn" ? "domain" : providers[name],
      url: "https://cdn.jsdelivr.net/gh/Loyalsoldier/clash-rules@release/" + file + ".txt",
      path: "./ruleset/" + name + ".yaml",
      interval: 86400
    };
  }

  config["rule-providers"]["my-direct"] = {
    type: "http",
    behavior: "domain",
    url: "https://gh-proxy.com/https://raw.githubusercontent.com/child9527/clash-latest/main/clash_rules/my-direct.yaml",
    path: "./ruleset/my-direct.yaml",
    interval: 86400
  };

  config["rule-providers"]["my-ai"] = {
    type: "http",
    behavior: "domain",
    url: "https://gh-proxy.com/https://raw.githubusercontent.com/child9527/clash-latest/main/clash_rules/my-ai.yaml",
    path: "./ruleset/my-ai.yaml",
    interval: 86400
  };

  // ============================================================
  // 5. 分流规则

  config.rules = [
    // 0. 基础设施直连
    ...infraDirectDomains.map(d => "DOMAIN-SUFFIX," + d + ",DIRECT"),
